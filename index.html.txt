    function initFirebase() {
        if (firebaseConfig && firebaseConfig.apiKey) {
            try {
                if (!firebase.apps.length) {
                    firebase.initializeApp(firebaseConfig);
                }
                dbRef = firebase.database().ref('futsal_matches');
                dbRef.on('value', (snapshot) => {
                    const val = snapshot.val();
                    if (val) {
                        matches = Object.values(val);
                        saveLocalOnly();
                        renderAll();
                    }
                });
            } catch(e) {
                console.log("Firebase error:", e);
            }
        }
    }

    // STATISTICHE & CLASSIFICA
    function calculateStats() {
        let stats = {};
        players.forEach(p => {
            stats[p] = { name: p, punti: 0, pg: 0, v: 0, n: 0, s: 0, gol: 0, autogol: 0, rigoriSbagliati: 0, totalScore: 0, votesReceived: 0 };
        });

        matches.forEach(m => {
            const isGialliWin = m.scoreGialli > m.scoreNeri;
            const isDraw = m.scoreGialli === m.scoreNeri;

            m.teamGialli.concat(m.teamNeri).forEach(p => {
                if (!stats[p]) stats[p] = { name: p, punti: 0, pg: 0, v: 0, n: 0, s: 0, gol: 0, autogol: 0, rigoriSbagliati: 0, totalScore: 0, votesReceived: 0 };
                stats[p].pg++;
            });

            m.teamGialli.forEach(p => {
                if (isGialliWin) { stats[p].punti += 3; stats[p].v++; }
                else if (isDraw) { stats[p].punti += 1; stats[p].n++; }
                else { stats[p].s++; }
            });

            m.teamNeri.forEach(p => {
                if (!isGialliWin && !isDraw) { stats[p].punti += 3; stats[p].v++; }
                else if (isDraw) { stats[p].punti += 1; stats[p].n++; }
                else { stats[p].s++; }
            });

            if (m.goals) {
                for (let [p, c] of Object.entries(m.goals)) { if (stats[p]) stats[p].gol += c; }
            }
            if (m.autogol) {
                for (let [p, c] of Object.entries(m.autogol)) { if (stats[p]) stats[p].autogol += c; }
            }
            if (m.rigoriSbagliati) {
                for (let [p, c] of Object.entries(m.rigoriSbagliati)) { if (stats[p]) stats[p].rigoriSbagliati += c; }
            }

            // Calcolo Pagelle
            if (m.ballots) {
                Object.values(m.ballots).forEach(ballot => {
                    Object.entries(ballot).forEach(([target, score]) => {
                        if (stats[target]) {
                            stats[target].totalScore += score;
                            stats[target].votesReceived += 1;
                        }
                    });
                });
            }
        });

        return Object.values(stats);
    }

    function renderClassifica() {
        const stats = calculateStats();
        const theadTr = document.getElementById('thead-tr');

        if (classificaSortMode === 'voti') {
            theadTr.innerHTML = `
                <th class="number">#</th>
                <th>Giocatore</th>
                <th class="number">Media Voto</th>
                <th class="number">Voti Ricevuti</th>
                <th class="number">PG</th>
            `;
            stats.sort((a, b) => {
                const avgA = a.votesReceived > 0 ? (a.totalScore / a.votesReceived) : 0;
                const avgB = b.votesReceived > 0 ? (b.totalScore / b.votesReceived) : 0;
                return avgB - avgA;
            });
        } else {
            theadTr.innerHTML = `
                <th class="number">#</th>
                <th>Giocatore</th>
                <th class="number">Punti</th>
                <th class="number">% Pti</th>
                <th class="number">PG</th>
                <th class="number">V</th>
                <th class="number">N</th>
                <th class="number">S</th>
                <th class="number">Gol</th>
                <th class="number">Gol/PG</th>
            `;
            if (classificaSortMode === 'punti') {
                stats.sort((a, b) => b.punti - a.punti || b.gol - a.gol || a.pg - b.pg);
            } else {
                stats.sort((a, b) => b.gol - a.gol || b.punti - a.punti);
            }
        }

        const tbody = document.getElementById('tbody-classifica');
        tbody.innerHTML = '';

        stats.forEach((p, index) => {
            if (p.pg === 0 && classificaSortMode !== 'voti') return;

            let rankClass = index === 0 ? 'rank-1' : index === 1 ? 'rank-2' : index === 2 ? 'rank-3' : '';
            const maxPunti = p.pg * 3;
            const percPunti = maxPunti > 0 ? ((p.punti / maxPunti) * 100).toFixed(0) + '%' : '0%';
            const golPerPg = p.pg > 0 ? (p.gol / p.pg).toFixed(2) : '0.00';
            const avgGrade = p.votesReceived > 0 ? (p.totalScore / p.votesReceived).toFixed(2) : 'N.V.';

            const tr = document.createElement('tr');
            if (classificaSortMode === 'voti') {
                tr.innerHTML = `
                    <td class="number"><span class="rank-pos ${rankClass}">${index + 1}</span></td>
                    <td><strong>${p.name}</strong></td>
                    <td class="number" style="color:var(--primary); font-weight:bold;">${avgGrade}</td>
                    <td class="number">${p.votesReceived}</td>
                    <td class="number">${p.pg}</td>
                `;
            } else {
                tr.innerHTML = `
                    <td class="number"><span class="rank-pos ${rankClass}">${index + 1}</span></td>
                    <td><strong>${p.name}</strong></td>
                    <td class="number"><strong>${p.punti}</strong></td>
                    <td class="number">${percPunti}</td>
                    <td class="number">${p.pg}</td>
                    <td class="number">${p.v}</td>
                    <td class="number">${p.n}</td>
                    <td class="number">${p.s}</td>
                    <td class="number"><strong>${p.gol}</strong></td>
                    <td class="number">${golPerPg}</td>
                `;
            }
            tbody.appendChild(tr);
        });
    }

            currentMatch.teamNeri = checked.slice(mid);

    function sorteggiaSquadre() {
        let shuffled = [...currentMatch.presenti].sort(() => Math.random() - 0.5);
        const mid = Math.ceil(shuffled.length / 2);
        currentMatch.teamGialli = shuffled.slice(0, mid);
        currentMatch.teamNeri = shuffled.slice(mid);
        renderLiveMatch();

    function setCapitano(player, team) {
        if (team === 'Gialli') {
            currentMatch.capitanoGialli = (currentMatch.capitanoGialli === player) ? null : player;
        } else {
            currentMatch.capitanoNeri = (currentMatch.capitanoNeri === player) ? null : player;
        }
        renderLiveMatch();
    }

    function renderLiveMatch() {
        const listGialli = document.getElementById('team-gialli-list');
        const listNeri = document.getElementById('team-neri-list');

        listGialli.innerHTML = '';
        listNeri.innerHTML = '';

        currentMatch.teamGialli.forEach(p => listGialli.appendChild(createPlayerMatchItem(p, 'Gialli')));
        currentMatch.teamNeri.forEach(p => listNeri.appendChild(createPlayerMatchItem(p, 'Neri')));
    }

    function createPlayerMatchItem(player, team) {
        const goals = currentMatch.goals[player] || 0;
        const ag = currentMatch.autogol[player] || 0;
        const rs = currentMatch.rigoriSbagliati[player] || 0;
        const isCap = (team === 'Gialli' && currentMatch.capitanoGialli === player) || (team === 'Neri' && currentMatch.capitanoNeri === player);

        const li = document.createElement('li');
        li.className = 'team-player-item';
        li.innerHTML = `
            <div class="player-row-top">
                <span>
                    <strong>${player}</strong>
                    ${isCap ? '<span class="captain-badge">👑 (C)</span>' : ''}
                </span>
                <button class="btn-ctrl btn-cap ${isCap ? 'is-cap' : ''}" onclick="setCapitano('${player}', '${team}')">Cap</button>
            </div>
            <div class="player-row-controls">
                <span style="font-size:0.7rem; color:#aaa; margin-right:auto;">
                    ${goals > 0 ? `⚽${goals} ` : ''}${ag > 0 ? `AG:${ag} ` : ''}${rs > 0 ? `RS:${rs}` : ''}
                </span>
                <button class="btn-ctrl" onclick="addEvent('${player}', 'goals', 1, '${team}')">+Gol</button>
                <button class="btn-ctrl" onclick="addEvent('${player}', 'goals', -1, '${team}')">-</button>
                <button class="btn-ctrl" onclick="addEvent('${player}', 'autogol', 1, '${team}')">+AG</button>
                <button class="btn-ctrl" onclick="addEvent('${player}', 'rigoriSbagliati', 1, '${team}')">+RS</button>
                <button class="btn-ctrl" onclick="spostaSquadra('${player}', '${team}')">⇄</button>
            </div>
        `;
        return li;
    }

    function addEvent(player, type, delta, team) {
        if (!currentMatch[type]) currentMatch[type] = {};
        currentMatch[type][player] = (currentMatch[type][player] || 0) + delta;
        if (currentMatch[type][player] <= 0) delete currentMatch[type][player];

        if (type === 'goals') {
            let input = document.getElementById(team === 'Gialli' ? 'score-gialli' : 'score-neri');
            input.value = Math.max(0, parseInt(input.value || 0) + delta);
        }

        if (type === 'autogol' && delta > 0) {
            let input = document.getElementById(team === 'Gialli' ? 'score-neri' : 'score-gialli');
            input.value = parseInt(input.value || 0) + 1;
        }

        renderLiveMatch();
    }

    function spostaSquadra(player, currentTeam) {
        if (currentTeam === 'Gialli') {
            currentMatch.teamGialli = currentMatch.teamGialli.filter(p => p !== player);
            currentMatch.teamNeri.push(player);
            if (currentMatch.capitanoGialli === player) currentMatch.capitanoGialli = null;
        } else {
            currentMatch.teamNeri = currentMatch.teamNeri.filter(p => p !== player);
            currentMatch.teamGialli.push(player);
            if (currentMatch.capitanoNeri === player) currentMatch.capitanoNeri = null;
        }
        renderLiveMatch();
    }

    function salvaPartita() {
        const matchDate = document.getElementById('match-date').value;
        const scoreG = parseInt(document.getElementById('score-gialli').value) || 0;
        const scoreN = parseInt(document.getElementById('score-neri').value) || 0;

        const existingMatch = matches.find(m => m.id === editingMatchId);
        const matchData = {
            id: editingMatchId || Date.now(),
            date: matchDate,
            scoreGialli: scoreG,
            scoreNeri: scoreN,
            capitanoGialli: currentMatch.capitanoGialli,
            capitanoNeri: currentMatch.capitanoNeri,
            teamGialli: [...currentMatch.teamGialli],
            teamNeri: [...currentMatch.teamNeri],
            goals: {...currentMatch.goals},
            autogol: {...currentMatch.autogol},
            rigoriSbagliati: {...currentMatch.rigoriSbagliati},
            ballots: existingMatch ? (existingMatch.ballots || {}) : {}
        };

        if (editingMatchId) {
            const idx = matches.findIndex(m => m.id === editingMatchId);
            if (idx !== -1) matches[idx] = matchData;
            editingMatchId = null;
        } else {
            matches.unshift(matchData);
        }

        saveData();
        alert("Partita salvata con successo!");
        
        currentMatch = { presenti: [], teamGialli: [], teamNeri: [], capitanoGialli: null, capitanoNeri: null, goals: {}, autogol: {}, rigoriSbagliati: {} };
        tornaASelezione();
        resetSelezione();
        switchTab('classifica');
    }

    function submitRatings(e) {
        e.preventDefault();
        const match = matches.find(m => m.id === activeVotingMatchId);
        if (!match) return;

        if (!match.ballots) match.ballots = {};

        const formData = new FormData(e.target);
        const ballot = {};

        for (let [key, val] of formData.entries()) {
            if (key.startsWith('rate-')) {
                const targetName = key.replace('rate-', '');
                ballot[targetName] = parseFloat(val);
            }
        }

        match.ballots[currentVoter] = ballot;
        saveData();

        alert("Voti salvati con successo!");
        document.getElementById('voting-step-form').style.display = 'none';
        document.getElementById('voting-step-identity').style.display = 'block';
        renderMatchReport(match);
    }

    function renderMatchReport(match) {
        const reportDiv = document.getElementById('report-list');
        const ballots = match.ballots || {};
        const countVotanti = Object.keys(ballots).length;
        document.getElementById('ballots-count-badge').innerText = `${countVotanti} Votanti`;

        const allPlayers = match.teamGialli.concat(match.teamNeri);
        const playerScores = {};

        allPlayers.forEach(p => playerScores[p] = { total: 0, count: 0 });

        Object.values(ballots).forEach(b => {
            Object.entries(b).forEach(([target, score]) => {
                if (playerScores[target]) {
                    playerScores[target].total += score;
                    playerScores[target].count += 1;
                }
            });
        });

        const sorted = allPlayers.map(p => {
            const c = playerScores[p].count;
            const avg = c > 0 ? (playerScores[p].total / c) : 0;
            return { name: p, avg, count: c };
        }).sort((a, b) => b.avg - a.avg);

        reportDiv.innerHTML = sorted.map((p, idx) => `
            <div style="display:flex; justify-content:space-between; align-items:center; padding:8px 0; border-bottom:1px solid #333;">
                <span>
                    <strong style="color:white;">${idx === 0 && p.avg > 0 ? '👑 ' : ''}${p.name}</strong> 
                    <small style="color:#777;">(${p.count} voti)</small>
                </span>
                <strong style="color:var(--primary); font-size:1rem;">${p.avg > 0 ? p.avg.toFixed(2) : 'N.V.'}</strong>
            </div>
        `).join('');

        document.getElementById('voting-step-report').style.display = 'block';
    }

    // STORICO PARTITE
    function renderStorico() {
        const container = document.getElementById('lista-storico');
        container.innerHTML = '';

        if (matches.length === 0) {
            container.innerHTML = '<div class="card">Nessuna partita registrata.</div>';
            return;
        }

        matches.forEach(m => {
            const card = document.createElement('div');
            card.className = 'card history-card';

            let eventsSummary = [];
            if (m.goals) {
                for (let [p, g] of Object.entries(m.goals)) eventsSummary.push(`${p} (${g}⚽)`);
            }
            if (m.autogol) {
                for (let [p, g] of Object.entries(m.autogol)) eventsSummary.push(`${p} (${g} AG)`);
            }

            const gialliList = m.teamGialli.map(p => p === m.capitanoGialli ? `${p} 👑` : p).join(', ');
            const neriList = m.teamNeri.map(p => p === m.capitanoNeri ? `${p} 👑` : p).join(', ');

            card.innerHTML = `
                <div class="card-title">
                    <span>Partita del ${m.date}</span>
                    <div>
                        <button class="btn btn-sm btn-secondary" onclick="modificaPartita(${m.id})"><i class="fas fa-edit"></i></button>
                        <button class="btn btn-sm btn-danger" onclick="eliminaPartita(${m.id})"><i class="fas fa-trash"></i></button>
                    </div>
                </div>
                <div class="score-board">
                    <span style="color: #fbc02d;">GIALLI ${m.scoreGialli}</span> - <span style="color: #aaaaaa;">${m.scoreNeri} NERI</span>
                </div>
                <div class="team-container" style="margin-bottom: 8px;">
                    <div style="font-size: 0.75rem;"><strong>GIALLI:</strong><br>${gialliList}</div>
                    <div style="font-size: 0.75rem;"><strong>NERI:</strong><br>${neriList}</div>
                </div>
                ${eventsSummary.length > 0 ? `
                    <div style="font-size: 0.78rem; color: #bbb; background: #262626; padding: 6px; border-radius: 4px; border:1px solid #333;">
                        <strong>Eventi:</strong> ${eventsSummary.join(', ')}
                    </div>
                ` : ''}
            `;
            container.appendChild(card);
        });
    }

    function eliminaPartita(id) {
        if (confirm("Eliminare questa partita dallo storico?")) {
            matches = matches.filter(m => m.id !== id);
            saveData();
        }
    }

    // GESTIONE GIOCATORI
    function renderGiocatoriList() {
        const ul = document.getElementById('lista-giocatori-full');
        ul.innerHTML = '';
        players.sort().forEach(p => {
            const li = document.createElement('li');
            li.className = 'team-player-item';
            li.style.flexDirection = 'row';
            li.style.justifyContent = 'space-between';
            li.innerHTML = `
                <span><strong>${p}</strong></span>
                <button class="btn btn-sm btn-danger" onclick="eliminaGiocatore('${p}')"><i class="fas fa-times"></i></button>
            `;
            ul.appendChild(li);
        });
    }

    function aggiungiGiocatore() {
        const input = document.getElementById('new-player-name');
        const name = input.value.trim();
        if (name && !players.includes(name)) {
            players.push(name);
            input.value = '';
            saveData();
        }
    }

    function eliminaGiocatore(name) {
        if (confirm(`Eliminare ${name}?`)) {
            players = players.filter(p => p !== name);
            saveData();
        }
    }

        window.print();
